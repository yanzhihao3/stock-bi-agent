# Alembic：表结构版本管理（"数据库的 Git"）。models/orm.py 里已经没有 create_all 了，
# 建表/改表全靠它 —— CI 的 "Apply migrations" 一步和 docker-compose 的启动命令
# 都直接调用 `python -m alembic upgrade head`，所以它是**直接依赖**，必须写在这里。
#
# ⚠️ 漏写这一行的翻车方式特别隐蔽：本机早就手动装过了，只有干净环境（CI、Docker
#    构建）才暴露。而且报错完全指不到依赖上：
#      No module named alembic.__main__; 'alembic' is a package and cannot be directly executed
#    这句话里说的"包"是仓库根目录那个**同名文件夹** `alembic/`（迁移脚本目录，
#    里面没有 __main__.py），真正的 alembic 根本没装上 —— 前半句像"装错了"，
#    后半句像"用错了"，两个方向都是错的，实际原因是"没装"。
#    排查技巧：把同一条命令换个目录执行。如果报错变成 `No module named alembic`
#    （少了 `.__main__` 那半句），就说明 cwd 参与了 import、本地目录顶了包。
#
# Requires-Python >=3.10，正好覆盖本项目的 3.10 / 3.11 两个环境。
alembic==1.19.2

# MySQL 驱动。选 pymysql 而不是 mysqlclient：纯 Python，Windows 上不用装编译器。
# 只有把 DATABASE_URL 指向 MySQL 时才会真正用到（默认走 SQLite）。
#
# ⚠️ 版本号要认 **dist-info**，别看 pymysql.__version__ —— 实测后者返回 '2.2.8'，
#    而这个包实际装的是 1.2.0（pymysql.VERSION == (1,2,0,'final')，
#    site-packages 下是 pymysql-1.2.0.dist-info），PyPI 上最高也只到 1.2.3。
#    按那个字符串 pin 会让 pip install -r requirements.txt 直接失败
#    （No matching distribution found），Docker / CI 全挂。踩过。
#
# [rsa] extra 会带上 cryptography：MySQL 8 的默认认证插件 caching_sha2_password
# 在**首次、未缓存**的连接上需要它，否则报：
#   RuntimeError: 'cryptography' package is required for sha256_password or
#                 caching_sha2_password auth methods
# 本机 root 用的是 mysql_native_password（查过 mysql.user），用不到它 —— 但换成
# Docker 里的 MySQL 8 / 新装服务器就会用到，而那个报错完全指不到依赖上，很难查。
# 所以默认带上。
PyMySQL[rsa]==1.2.0
# cryptography 必须**显式 pin**：不写的话 pip 会在构建时现解析最新版，
# Docker 镜像就不可复现了（违反本文件开头的约定）。
# 50.0.1 是本机实测装上的版本，也是 AutoDL 那次部署环境里的版本
# （见 docs/requirements.lock-autodl-20260913.txt）。
cryptography==50.0.1
redis==6.4.0
